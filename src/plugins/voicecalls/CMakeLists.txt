project(voicecalls)

set(PLUGIN_NAME "voicecalls")
set(PLUGIN_DISPLAY_NAME "Voice Calls")
set(PLUGIN_DESCRIPTION "Allows to make and receive voice calls")
set(PLUGIN_DEPENDENCIES xmppstreams stanzaprocessor jingle)

# AUTO preserves optional source builds; release presets explicitly require ON.
set(PLUGIN_voicecalls AUTO CACHE STRING "Build voice calls: AUTO, ON or OFF")
set_property(CACHE PLUGIN_voicecalls PROPERTY STRINGS AUTO ON OFF)
string(TOUPPER "${PLUGIN_voicecalls}" _voicecalls_mode)
if(NOT _voicecalls_mode MATCHES "^(AUTO|ON|YES|TRUE|1|OFF|NO|FALSE|0)$")
    message(FATAL_ERROR "PLUGIN_voicecalls must be AUTO, ON or OFF")
endif()
if(NOT PLUGIN_voicecalls)
    return()
endif()

if(WIN32)
    set(VACUUM_GSTREAMER_ROOT "$ENV{GSTREAMER_1_0_ROOT_MSVC_X86_64}" CACHE PATH
        "GStreamer MSVC runtime and development installation prefix")
    if(VACUUM_GSTREAMER_ROOT)
        file(TO_CMAKE_PATH "${VACUUM_GSTREAMER_ROOT}" VACUUM_GSTREAMER_ROOT)
        set(ENV{PKG_CONFIG_PATH} "${VACUUM_GSTREAMER_ROOT}/lib/pkgconfig;$ENV{PKG_CONFIG_PATH}")
        set(PKG_CONFIG_ARGN "--define-variable=prefix=${VACUUM_GSTREAMER_ROOT}")
        list(PREPEND CMAKE_PROGRAM_PATH "${VACUUM_GSTREAMER_ROOT}/bin")
    endif()
endif()
find_package(PkgConfig QUIET)
if(PKG_CONFIG_FOUND)
    pkg_check_modules(VOICECALLS_GST QUIET IMPORTED_TARGET
        gstreamer-1.0>=1.20 gstreamer-sdp-1.0 gstreamer-webrtc-1.0)
endif()
if(NOT VOICECALLS_GST_FOUND)
    if(_voicecalls_mode STREQUAL AUTO)
        message(STATUS "GStreamer WebRTC development files not found; voicecalls is disabled (AUTO)")
        return()
    endif()
    message(FATAL_ERROR "Voice calls were requested, but GStreamer SDP/WebRTC development files are missing")
endif()

set_property(GLOBAL PROPERTY VACUUM_GSTREAMER_VERSION "${VOICECALLS_GST_gstreamer-1.0_VERSION}")
if(WIN32 AND NOT VACUUM_GSTREAMER_ROOT)
    pkg_get_variable(VACUUM_GSTREAMER_ROOT gstreamer-1.0 prefix)
endif()
set_property(GLOBAL PROPERTY VACUUM_GSTREAMER_ROOT "${VACUUM_GSTREAMER_ROOT}")
include("voicecalls.cmake")
set(ADD_LIBS PkgConfig::VOICECALLS_GST)
include("${CMAKE_SOURCE_DIR}/src/plugins/plugins.cmake")
if(IS_ENABLED)
    target_link_libraries(${PLUGIN_NAME} Qt6::Multimedia)
endif()
